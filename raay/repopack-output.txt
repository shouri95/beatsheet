// src/components/Playground/PlaygroundCanvas.tsx

"use client"

import React, { useCallback } from 'react'
import { useSelector, useDispatch } from 'react-redux'
import { RootState } from '../../store'
import ReactFlow, {
  Node,
  Edge,
  Connection,
  addEdge,
  Background,
  Controls,
  MiniMap,
  useNodesState,
  useEdgesState,
  ConnectionLineType,
} from 'reactflow'
import 'reactflow/dist/style.css'
import CustomSceneNode from './CustomSceneNode'
import { addConnection } from '../../store/connectionsSlice'

const nodeTypes = {
  scene: CustomSceneNode,
}

const PlaygroundCanvas: React.FC = () => {
  const dispatch = useDispatch()
  const scenes = useSelector((state: RootState) => state.scenes.scenes)
  const connections = useSelector((state: RootState) => state.connections.connections)

  const [nodes, setNodes, onNodesChange] = useNodesState(
    scenes.map((scene) => ({
      id: scene.id,
      type: 'scene',
      data: { scene },
      position: { x: scene.position.x, y: scene.position.y },
    }))
  )

  const [edges, setEdges, onEdgesChange] = useEdgesState(
    connections.map((conn) => ({
      id: `e${conn.from}-${conn.to}`,
      source: conn.from,
      target: conn.to,
      animated: true,
      type: ConnectionLineType.SmoothStep,
    }))
  )

  const onConnect = useCallback(
    (params: Connection) => {
      setEdges((eds) => addEdge(params, eds))
      if (params.source && params.target) {
        dispatch(addConnection({ from: params.source, to: params.target }))
      }
    },
    [dispatch, setEdges]
  )

  return (
    <div className="h-full">
      <ReactFlow
        nodes={nodes}
        edges={edges}
        onNodesChange={onNodesChange}
        onEdgesChange={onEdgesChange}
        onConnect={onConnect}
        nodeTypes={nodeTypes}
        snapToGrid={true}
        snapGrid={[15, 15]}
        fitView
      >
        <Background color="#e0e0e0" gap={16} />
        <Controls />
        <MiniMap />
      </ReactFlow>
    </div>
  )
}

export default PlaygroundCanvas

import React, { useRef, useEffect } from 'react';
import { useDrag } from 'react-dnd';
import { Card } from '../Common/Card';

interface PropItemProps {
  prop: {
    id: string;
    name: string;
    description: string;
  };
}

interface DragItem {
  id: string;
  type: string;
}

const PropItem: React.FC<PropItemProps> = ({ prop }) => {
  const ref = useRef<HTMLDivElement>(null);

  const [{ isDragging }, drag] = useDrag<DragItem, void, { isDragging: boolean }>(() => ({
    type: 'PROP',
    item: { id: prop.id, type: 'PROP' },
    collect: (monitor) => ({
      isDragging: monitor.isDragging(),
    }),
  }));

  useEffect(() => {
    if (ref.current) {
      drag(ref.current);
    }
  }, [drag]);

  return (
    <div
      ref={ref}
      className={`bg-green-100 p-2 rounded-md mb-2 cursor-pointer ${
        isDragging ? 'opacity-50' : 'opacity-100'
      }`}
    >
      <h5 className="text-sm font-medium">{prop.name}</h5>
    </div>
  );
};

export default PropItem;

import React, { useState } from 'react'
import { Handle, Position } from 'reactflow'

import { RootState } from '../../store'

  data: {
    scene: {
      id: string
      title: string
      scriptContent: string
      position: { x: number; y: number }
      size: { width: number; height: number }
      characters: string[]
      props: string[]
      color: string
    }
  }

const SceneBox: React.FC<SceneBoxProps> = ({ data }) => {
  const dispatch = useDispatch()
  const { scene } = data
  const characters = useSelector((state: RootState) =>
    state.characters.characters.filter(char => scene.characters.includes(char.id))
  )

    state.props.props.filter(prop => scene.props.includes(prop.id))

// src/components/Playground/Toolbar.tsx

"use client"

import React from 'react'
import { Button } from '../Common/Button'
import { useDispatch, useSelector } from 'react-redux'
import { addScene } from '../../store/scenesSlice'
import { addCharacter } from '../../store/charactersSlice'
import { addProp } from '../../store/propsSlice'
import { v4 as uuidv4 } from 'uuid'
import { UserCircleIcon, PlusCircleIcon } from '@heroicons/react/24/outline'
import { Menu, Transition } from '@headlessui/react'
import { Fragment } from 'react'
import { RootState } from '@/store'

const Toolbar: React.FC = () => {
  const dispatch = useDispatch()
  const scenes = useSelector((state: RootState) => state.scenes.scenes)

  const handleAddScene = () => {
    const newSceneNumber = scenes.length + 1
    const newScene = {
      id: uuidv4(),
      title: `Scene ${newSceneNumber}`,
      scriptContent: '',
      position: getNextScenePosition(scenes),
      size: { width: 200, height: 150 },
      characters: [],
      props: [],
      color: getRandomColor(),
    }
    dispatch(addScene(newScene))
  }

  const getNextScenePosition = (existingScenes: any[]): { x: number; y: number } => {
    // Simple logic to position new scenes incrementally
    if (existingScenes.length === 0) {
      return { x: 100, y: 100 }
    }
    const lastScene = existingScenes[existingScenes.length - 1]
    return { x: lastScene.position.x + 50, y: lastScene.position.y + 50 }
  }

  const getRandomColor = (): string => {
    const colors = ['bg-blue-100', 'bg-green-100', 'bg-yellow-100', 'bg-red-100', 'bg-purple-100']
    return colors[Math.floor(Math.random() * colors.length)]
  }

  const handleAddCharacter = () => {
    const newCharacter = {
      id: uuidv4(),
      name: `Character ${uuidv4().slice(0, 4)}`,
      description: '',
    }
    dispatch(addCharacter(newCharacter))
  }

  const handleAddProp = () => {

"use client"

import React, { Component, ErrorInfo, ReactNode } from 'react'

interface Props {
  children: ReactNode
}

interface State {
  hasError: boolean
}

class ErrorBoundary extends Component<Props, State> {
  state: State = {
    hasError: false,
  }

  static getDerivedStateFromError(_: Error): State {
    return { hasError: true }
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error("Uncaught error:", error, errorInfo)
  }

  render() {
    if (this.state.hasError) {
      return <h1 className="text-center mt-20 text-2xl">Something went wrong.</h1>
    }

    return this.props.children
  }
}

export default ErrorBoundary

import { useState } from 'react'

const useZoomPan = () => {
  const [zoom, setZoom] = useState(1)
  const [pan, setPan] = useState({ x: 0, y: 0 })

  const zoomIn = () => setZoom(prev => prev + 0.1)
  const zoomOut = () => setZoom(prev => prev - 0.1)
  const panMove = (x: number, y: number) => setPan({ x, y })

  return { zoom, pan, zoomIn, zoomOut, panMove }
}

// src/store/scenesSlice.ts

import { createSlice, PayloadAction } from '@reduxjs/toolkit'

interface Scene {
  id: string
  title: string
  scriptContent: string
  position: { x: number; y: number }
  size: { width: number; height: number }
  characters: string[]
  props: string[]
  color: string
}

interface ScenesState {
  scenes: Scene[]
}

const initialState: ScenesState = {
  scenes: [],
}

const scenesSlice = createSlice({
  name: 'scenes',
  initialState,
  reducers: {
    addScene: (state, action: PayloadAction<Scene>) => {
      state.scenes.push(action.payload)
    },
    updateScenePosition: (
      state,
      action: PayloadAction<{ id: string; position: { x: number; y: number } }>
    ) => {
      const scene = state.scenes.find(scene => scene.id === action.payload.id)
      if (scene) {
        scene.position = action.payload.position
      }
    },
    updateScene: (
      state,
      action: PayloadAction<{ id: string; title: string; scriptContent: string }>
    ) => {
      const scene = state.scenes.find(scene => scene.id === action.payload.id)
      if (scene) {
        scene.title = action.payload.title
        scene.scriptContent = action.payload.scriptContent
      }
    },
    assignCharacterToScene: (
      state,
      action: PayloadAction<{ sceneId: string; characterId: string }>
    ) => {
      const scene = state.scenes.find(scene => scene.id === action.payload.sceneId)
      if (scene && !scene.characters.includes(action.payload.characterId)) {
        scene.characters.push(action.payload.characterId)
      }
    },
    assignPropToScene: (
      state,
      action: PayloadAction<{ sceneId: string; propId: string }>
    ) => {
      const scene = state.scenes.find(scene => scene.id === action.payload.sceneId)
      if (scene && !scene.props.includes(action.payload.propId)) {
        scene.props.push(action.payload.propId)
      }
    },
    // Optional: Remove Scene, etc.
  },
})

export const {
  addScene,
  updateScenePosition,
  updateScene,
  assignCharacterToScene,
  assignPropToScene,
} = scenesSlice.actions
export default scenesSlice.reducer
