        outline: 'border border-gray-300 bg-transparent hover:bg-gray-100 dark:border-gray-600 dark:bg-transparent dark:hover:bg-gray-700',
        ghost: 'text-gray-900 dark:text-white hover:bg-gray-100 dark:hover:bg-gray-800',
        link: 'text-primary underline-offset-4 hover:underline',
      },
      size: {
        default: 'h-10 px-4 py-2',
        sm: 'h-8 rounded-full px-3',
        lg: 'h-12 rounded-full px-6',
        icon: 'h-10 w-10',
      },
    },
    defaultVariants: {
      variant: 'default',
      size: 'default',
    },
  }
);

export interface ButtonProps
  extends React.ButtonHTMLAttributes<HTMLButtonElement>,
    VariantProps<typeof buttonVariants> {
  asChild?: boolean;
}

const Button = React.forwardRef<HTMLButtonElement, ButtonProps>(
  ({ className, variant, size, asChild = false, ...props }, ref) => {
    const Comp = asChild ? Slot : 'button';
    return (
      <Comp
        className={cn(buttonVariants({ variant, size, className }))}
        ref={ref}
        {...props}
      />
    );
  }
);
Button.displayName = 'Button';

import { useState } from 'react';

const ThemeToggle = () => {
  const [theme, setTheme] = useState<'light' | 'dark'>('light');

  const toggleTheme = () => {
    const newTheme = theme === 'light' ? 'dark' : 'light';
    document.documentElement.setAttribute('data-theme', newTheme);
    setTheme(newTheme);
  };

  return (
    <Button variant="outline" size="sm" onClick={toggleTheme}>
      Toggle Theme
    </Button>
  );
};

import React from 'react';
import { Scene } from "../types";
import { motion } from 'framer-motion';
import { Card, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { ChevronRight, ChevronLeft } from 'lucide-react';

type AdvancedTimelineViewProps = {
  scenes: Scene[];
  updateScenes: (scenes: Scene[]) => void;
  openDetails: (scene: Scene) => void;
};

const AdvancedTimelineView: React.FC<AdvancedTimelineViewProps> = ({ scenes, updateScenes, openDetails }) => {
  const [currentIndex, setCurrentIndex] = React.useState(0);

  const nextScene = () => {
    if (currentIndex < scenes.length - 1) {
      setCurrentIndex(currentIndex + 1);
    }
  };

  const prevScene = () => {
    if (currentIndex > 0) {
      setCurrentIndex(currentIndex - 1);
    }
  };

  const getSceneColor = (scene: Scene) => {
    if (scene.duration > 5) return 'bg-red-200 dark:bg-red-800';
    if (scene.notes.includes('emotional')) return 'bg-yellow-200 dark:bg-yellow-800';
    if (scene.notes.includes('dialogue')) return 'bg-blue-200 dark:bg-blue-800';
    return 'bg-gray-200 dark:bg-gray-800';
  };

  return (
    <div className="relative w-full py-12">
      <div className="absolute left-0 right-0 top-1/2 h-1 bg-gray-300 dark:bg-gray-700 transform -translate-y-1/2" />
      <div className="flex justify-between items-center">
        <Button onClick={prevScene} disabled={currentIndex === 0} size="icon" variant="outline">
          <ChevronLeft className="h-4 w-4" />
        </Button>
        <div className="flex-1 px-4">
          <motion.div
            className="flex"
            animate={{ x: `${-currentIndex * 100}%` }}
            transition={{ type: 'spring', stiffness: 300, damping: 30 }}
          >
            {scenes.map((scene, index) => (
              <Card
                key={scene.id}
                className={`flex-shrink-0 w-full mx-2 cursor-pointer transition-all duration-300 ${
                  index === currentIndex ? 'scale-105 shadow-lg' : 'scale-95 opacity-50'
                } ${getSceneColor(scene)}`}
                onClick={() => openDetails(scene)}
              >
                <CardContent className="p-4">
                  <h3 className="font-bold text-lg mb-2">{scene.title}</h3>
                  <p className="text-sm mb-2">{scene.content.substring(0, 100)}...</p>
                  <p className="text-xs text-gray-500 dark:text-gray-400">Duration: {scene.duration} mins</p>
                </CardContent>
              </Card>
            ))}
          </motion.div>
        </div>
        <Button onClick={nextScene} disabled={currentIndex === scenes.length - 1} size="icon" variant="outline">
          <ChevronRight className="h-4 w-4" />
        </Button>
      </div>
    </div>
  );
};

export default AdvancedTimelineView;

const StoryboardComponent: React.FC = () => {
  const [scenes, setScenes] = useState<Scene[]>(initialScenes);
  const [draggedScene, setDraggedScene] = useState<Scene | null>(null);
  const [selectedScene, setSelectedScene] = useState<Scene | null>(null);
  const [history, setHistory] = useState<Scene[][]>([initialScenes]);
  const [historyIndex, setHistoryIndex] = useState(0);


  const onDragStart = (e: React.DragEvent<HTMLDivElement>, scene: Scene) => {
    setDraggedScene(scene);
  };

  const onDragOver = (e: React.DragEvent<HTMLDivElement>) => {
    e.preventDefault();
  };

  const onDrop = (e: React.DragEvent<HTMLDivElement>, targetIndex: number) => {
    e.preventDefault();
    if (draggedScene) {
      const newScenes = scenes.filter(scene => scene.id !== draggedScene.id);
      newScenes.splice(targetIndex, 0, draggedScene);
      setScenes(newScenes);
      saveToHistory(newScenes);
    }
    setDraggedScene(null);
  };

  const removeScene = (sceneId: number) => {
    const newScenes = scenes.filter(scene => scene.id !== sceneId);
    setScenes(newScenes);
    saveToHistory(newScenes);
  };

  const updateScene = (updatedScene: Scene) => {
    const newScenes = scenes.map(scene => scene.id === updatedScene.id ? updatedScene : scene);
    setScenes(newScenes);
    saveToHistory(newScenes);
  };

  const openDetails = (scene: Scene) => {
    setSelectedScene(scene);
  };

  const closeDetails = () => {
    setSelectedScene(null);
  };

  const saveToHistory = (newScenes: Scene[]) => {
    const newHistory = [...history.slice(0, historyIndex + 1), newScenes];
    setHistory(newHistory);
    setHistoryIndex(newHistory.length - 1);
  };

  const undo = () => {
    if (historyIndex > 0) {
      setHistoryIndex(historyIndex - 1);
      setScenes(history[historyIndex - 1]);
    }
  };

  const redo = () => {
    if (historyIndex < history.length - 1) {
      setHistoryIndex(historyIndex + 1);
      setScenes(history[historyIndex + 1]);
    }
  };

  const totalDuration = scenes.reduce((sum, scene) => sum + (scene.duration || 0), 0);

  const addNewScene = () => {
    const newScene: Scene = {
      id: scenes.length + 1,
      title: `New Scene ${scenes.length + 1}`,
      content: "Add your scene content here...",
      duration: 0,
      notes: '',
      transition: ''
    };
    const newScenes = [...scenes, newScene];
    setScenes(newScenes);
    saveToHistory(newScenes);
  };

  const totalDuration = scenes.reduce((sum, scene) => sum + (scene.duration || 0), 0);

  return (
    <div className="bg-gray-50 dark:bg-gray-900 min-h-screen p-6">
      <h1 className="text-3xl font-bold mb-8 text-center text-gray-900 dark:text-white">Interactive Storyboard</h1>
      
      <Tabs defaultValue="storyboard" className="w-full mb-8">
        <TabsList className="grid w-full grid-cols-3">
          <TabsTrigger value="storyboard">Storyboard</TabsTrigger>
          <TabsTrigger value="timeline">Timeline</TabsTrigger>
          <TabsTrigger value="script">Script</TabsTrigger>
        </TabsList>
        <TabsContent value="storyboard">
          <div className="flex flex-wrap justify-center gap-6 overflow-x-auto pb-8">
            {scenes.map((scene, index) => (
              <SceneCard
                key={scene.id}
                scene={scene}
                index={index}
                onDragStart={onDragStart}
                onDragOver={onDragOver}
                onDrop={onDrop}
                removeScene={removeScene}
                openDetails={openDetails}
              />
            ))}
            <Card className="flex items-center justify-center w-64 h-64 cursor-pointer hover:bg-gray-100 dark:hover:bg-gray-800 transition-colors" onClick={addNewScene}>
              <CardContent>
                <PlusCircle className="w-12 h-12 text-gray-400" />
                <p className="mt-2 text-gray-500">Add New Scene</p>
              </CardContent>
            </Card>
          </div>
        </TabsContent>
        <TabsContent value="timeline">
          <AdvancedTimelineView scenes={scenes} updateScenes={setScenes} openDetails={openDetails} />
        </TabsContent>
        <TabsContent value="script">
          <ScriptEditor />
        </TabsContent>
      </Tabs>
      
      {selectedScene && (
        <SceneDetailsPanel
          scene={selectedScene}
          isOpen={!!selectedScene}
          onClose={closeDetails}
          updateScene={updateScene}
        />
      )}
      
      <div className="flex justify-between items-center mt-4">
        <div className="flex space-x-2">
          <Button onClick={undo} disabled={historyIndex === 0} variant="outline" size="icon">
            <Undo className="h-4 w-4" />
          </Button>
          <Button onClick={redo} disabled={historyIndex === history.length - 1} variant="outline" size="icon">
            <Redo className="h-4 w-4" />
          </Button>
        </div>
        <p className="text-gray-700 dark:text-gray-300">Total Duration: {totalDuration} mins</p>
      </div>
    </div>
  );
};

export default StoryboardComponent;
